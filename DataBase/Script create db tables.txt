-- =================================================================
-- SCRIPT DE CRÉATION DE SCHÉMA POSTGRESQL (NEON DB)
-- Application de Camping et Marketplace
-- =================================================================

-- Suppression des tables existantes (pour un re-run facile du script)
DROP TABLE IF EXISTS event_participants;
DROP TABLE IF EXISTS reviews;
DROP TABLE IF EXISTS rental_items;
DROP TABLE IF EXISTS products;
DROP TABLE IF EXISTS reservations;
DROP TABLE IF EXISTS events;
DROP TABLE IF EXISTS centers;
DROP TABLE IF EXISTS payments;
DROP TABLE IF EXISTS cart;
DROP TABLE IF EXISTS users;

-- =================================================================
-- 1. MODULE DE GESTION DES UTILISATEURS (USERS)
-- =================================================================

CREATE TABLE "users" (
    user_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    phone VARCHAR(20),
    password VARCHAR(250),
    preferences JSONB DEFAULT '{}',
    user_type VARCHAR(20) NOT NULL DEFAULT 'user',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
)

-- =================================================================
-- 2. MODULE DE RÉSERVATION ET PAIEMENT (CENTERS)
-- (Nécessaire avant les réservations et autres éléments publics)
-- =================================================================

CREATE TABLE centers (
    center_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    -- Clé étrangère vers l'utilisateur propriétaire du centre
    owner_id UUID NOT NULL REFERENCES users(user_id) ON DELETE RESTRICT,
    name VARCHAR(150) NOT NULL,
    description TEXT,
    -- Stockage des coordonnées géographiques (Lat/Lon)
    latitude NUMERIC(9, 6),
    longitude NUMERIC(9, 6),
    -- L'adresse peut être un objet JSONB pour les détails (rue, ville, code postal)
    address JSONB DEFAULT '{}',
    amenities TEXT[], -- Tableau de chaînes (e.g., ['wifi', 'piscine', 'douche'])
    -- Tarifs de base stockés en JSONB (e.g., { "tent": 30.00, "RV": 50.00, "cabin": 80.00 })
    daily_rates JSONB DEFAULT '{}',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- =================================================================
-- 3. MODULE DE GESTION DES ÉQUIPEMENTS (LOCATION / ACHAT)
-- (Catalogue des produits publics)
-- =================================================================

CREATE TABLE rental_items (
    rental_item_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    -- L'équipement est lié à un centre spécifique
    center_id UUID NOT NULL REFERENCES centers(center_id) ON DELETE CASCADE,
    name VARCHAR(100) NOT NULL,
    description TEXT,
    daily_rate NUMERIC(10, 2) NOT NULL,
    stock_quantity INTEGER NOT NULL CHECK (stock_quantity >= 0),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE products (
    product_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    -- Le vendeur peut être un utilisateur ou un propriétaire de centre
    seller_id UUID NOT NULL REFERENCES users(user_id) ON DELETE RESTRICT,
    -- Optionnel: Si le produit est vendu par un centre (un magasin physique dans le centre)
    center_id UUID REFERENCES centers(center_id) ON DELETE SET NULL,
    name VARCHAR(150) NOT NULL,
    description TEXT,
    price NUMERIC(10, 2) NOT NULL CHECK (price > 0),
    -- 'neuf', 'occasion', 'reconditionné'
    condition VARCHAR(50) NOT NULL,
    category VARCHAR(50),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- =================================================================
-- 2. MODULE DE RÉSERVATION ET PAIEMENT (SUITE)
-- =================================================================

CREATE TABLE reservations (
    reservation_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    center_id UUID NOT NULL REFERENCES centers(center_id) ON DELETE RESTRICT,
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    -- Type d'emplacement réservé ('tent', 'RV', 'small_tent', etc.)
    emplacement_type VARCHAR(50) NOT NULL,
    total_price NUMERIC(10, 2) NOT NULL CHECK (total_price >= 0),
    -- 'confirmed', 'pending', 'cancelled', 'completed'
    status VARCHAR(20) NOT NULL DEFAULT 'pending',
    booking_details JSONB, -- Pour stocker les détails supplémentaires de la réservation
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    
    -- Contrainte pour garantir que la date de fin est après la date de début
    CHECK (end_date > start_date)
);

-- TABLE DE LIAISON pour les articles dans une réservation (location ou achat)
CREATE TABLE reservation_items (
    reservation_item_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    reservation_id UUID NOT NULL REFERENCES reservations(reservation_id) ON DELETE CASCADE,
    -- Identifiant de l'article (soit un rental_item, soit un product)
    item_id UUID NOT NULL,
    item_type VARCHAR(20) NOT NULL, -- 'rental' ou 'product'
    quantity INTEGER NOT NULL CHECK (quantity > 0),
    unit_price NUMERIC(10, 2) NOT NULL,
    
    -- Contrainte pour s'assurer que le type d'article est valide
    CHECK (item_type IN ('rental', 'product'))
);

CREATE TABLE payments (
    payment_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    -- Paiement lié à une réservation ou à un achat direct (optional)
    reservation_id UUID REFERENCES reservations(reservation_id) ON DELETE SET NULL,
    amount NUMERIC(10, 2) NOT NULL CHECK (amount > 0),
    currency CHAR(3) NOT NULL DEFAULT 'EUR',
    payment_method VARCHAR(50),
    -- 'succeeded', 'failed', 'refunded'
    status VARCHAR(20) NOT NULL,
    transaction_ref VARCHAR(100), -- Référence externe (Stripe, Paypal...)
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- TABLE DU PANIER (Gestion simple avec JSONB, car elle est transitoire)
CREATE TABLE cart (
    user_id UUID PRIMARY KEY REFERENCES users(user_id) ON DELETE CASCADE,
    -- Stocke la structure complète du panier
    items JSONB NOT NULL DEFAULT '[]',
    last_updated TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- =================================================================
-- 4. MODULE MARKETPLACE (AVIS)
-- =================================================================

CREATE TABLE reviews (
    review_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(user_id) ON DELETE RESTRICT,
    -- Clés optionnelles : la revue concerne soit un centre, soit un produit
    center_id UUID REFERENCES centers(center_id) ON DELETE CASCADE,
    product_id UUID REFERENCES products(product_id) ON DELETE CASCADE,
    -- 'center' ou 'product'
    entity_type VARCHAR(10) NOT NULL,
    rating INTEGER NOT NULL CHECK (rating BETWEEN 1 AND 5),
    comment TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    
    -- Contrainte pour s'assurer qu'une seule entité est notée
    CHECK (
        (center_id IS NOT NULL AND product_id IS NULL AND entity_type = 'center') OR
        (center_id IS NULL AND product_id IS NOT NULL AND entity_type = 'product')
    )
);

-- =================================================================
-- 5. MODULE ÉVÉNEMENTS ET COMMUNAUTÉ
-- =================================================================

CREATE TABLE events (
    event_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    creator_id UUID NOT NULL REFERENCES users(user_id) ON DELETE RESTRICT,
    -- Optionnel: Si l'événement est organisé par un centre
    center_id UUID REFERENCES centers(center_id) ON DELETE SET NULL,
    title VARCHAR(200) NOT NULL,
    description TEXT,
    event_date TIMESTAMP WITH TIME ZONE NOT NULL,
    location JSONB, -- Pour stocker l'adresse ou les coordonnées de l'événement
    max_participants INTEGER CHECK (max_participants IS NULL OR max_participants > 0),
    is_public BOOLEAN NOT NULL DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE event_participants (
    event_id UUID NOT NULL REFERENCES events(event_id) ON DELETE CASCADE,
    user_id UUID NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    joined_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    
    PRIMARY KEY (event_id, user_id) -- Clé composite pour éviter les doublons
);

-- =================================================================
-- INDEXATION POUR LA PERFORMANCE
-- =================================================================

CREATE INDEX idx_centers_owner_id ON centers(owner_id);
CREATE INDEX idx_reservations_user_id ON reservations(user_id);
CREATE INDEX idx_reservations_center_dates ON reservations(center_id, start_date, end_date);
CREATE INDEX idx_products_seller_id ON products(seller_id);
CREATE INDEX idx_reviews_entity ON reviews(entity_type, center_id, product_id);
CREATE INDEX idx_events_date ON events(event_date);

CREATE TABLE admLicense (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    activatedDate DATE,
    expireDate DATE,
    userId UUID REFERENCES users(user_id),
    userNumber INTEGER,
    activeUserNumber INTEGER,
    Description TEXT
);
